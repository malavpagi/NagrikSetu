│   │
│   ├── models/                # Mongoose Schemas (Section 43 & 44)

│   │
│   └── utils/                 # Helper functions
│       ├── geoUtils.js        # Distance calculation helpers
│       └── asyncWrapper.js    # Try-catch wrapper for async controllers

├── uploads/                   # Local folder for Multer to store images
